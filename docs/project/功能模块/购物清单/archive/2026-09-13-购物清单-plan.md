@dataclass

    """清单单条记录"""
    id: str          # uuid 前 8 位
    content: str     # 需求原文（自由文本）
    created_at: str  # "YYYY-MM-DD HH:MM:SS"

    def __init__(self, file_path: str = LIST_FILE):

        self.items: List[ListItem] = self._load()

    def _load(self) -> List[ListItem]:
        if os.path.exists(self.file_path):
            try:
                with open(self.file_path, "r", encoding="utf-8") as f:
                    data = json.load(f)
                items = []
                for d in data:
                    if "id" not in d or not d["id"]:
                        d["id"] = str(uuid.uuid4())[:8]
                    items.append(ListItem(**d))
                return items
            except Exception:
                return []
        return []

    def _save(self) -> None:
        with open(self.file_path, "w", encoding="utf-8") as f:
            json.dump([it.to_dict() for it in self.items], f, ensure_ascii=False, indent=2)

    def clear(self) -> str:
        raise NotImplementedError

        raise NotImplementedError

    def list_text(self) -> str:
        raise NotImplementedError

    def to_list(self) -> List[Dict[str, Any]]:

```python
    def add(self, content: str) -> str:
        """添加需求；与现有待处理条目完全同文则不重复添加"""
        content = (content or "").strip()
        if not content:
            return "需求内容不能为空。用法：`记到清单：想买的东西或想解决的问题`"
        for i, it in enumerate(self.items, 1):
            if not it.done and it.content == content:
                return f"清单里已有这条待处理需求（第{i}项）：{content}"
        self.items.append(ListItem(id=str(uuid.uuid4())[:8], content=content,
                                   created_at=time.strftime("%Y-%m-%d %H:%M:%S")))
        self._save()
        return (f"已记到清单（第{len(self.items)}项）：{content}\n"
                "想开始选购时，到侧边栏「购物清单」页点该条的「去推荐」。")

    def remove(self, index: int) -> str:
        if 1 <= index <= len(self.items):
            it = self.items.pop(index - 1)
            self._save()
            return f"已从清单移除第{index}项：{it.content}"
        return f"清单没有第{index}项（当前共{len(self.items)}项）"

    def toggle_done(self, index: int) -> str:
        if 1 <= index <= len(self.items):
            it = self.items[index - 1]
            it.done = not it.done
            self._save()
            return (f"第{index}项「{it.content}」已标记完成。" if it.done
                    else f"第{index}项「{it.content}」已改回待处理。")
        return f"清单没有第{index}项"

    def clear_done(self) -> str:
        n = sum(1 for it in self.items if it.done)
        self.items = [it for it in self.items if not it.done]
        self._save()
        return f"已清空{n}项已完成条目。"

    def clear(self) -> str:
        n = len(self.items)
        self.items.clear()
        self._save()
        return f"已清空购物清单（共{n}项）。"

    def pending_count(self) -> int:
        return sum(1 for it in self.items if not it.done)

    def list_text(self) -> str:
        if not self.items:
            return ("购物清单为空。想到想买的随时记下来：\n"
                    "· 对话里说 `记到清单：xxx`\n"
                    "· 或在侧边栏「购物清单」页添加")
        lines = ["**我的购物清单**（想买/想解决的需求池；清单收藏的是需求，转成对话后商品收藏进购物车）",
                 "| # | 需求 | 状态 | 记录时间 |",
                 "|---|---|---|---|"]
        for i, it in enumerate(self.items, 1):
            state = "已完成" if it.done else "待处理"
            lines.append(f"| {i} | {it.content} | {state} | {it.created_at} |")
        lines.append("")
        lines.append("指令：`记到清单：xxx` / `完成第N项` / `从清单移除第N项`；编号即存储顺序。")
        return "\n".join(lines)

    def to_list(self) -> List[Dict[str, Any]]:
        return [it.to_dict() for it in self.items]
```

    def _api_shopping_list_post(self, body: dict):
        """
        action:
          - add         {content}
          - remove      {index}
          - toggle_done {index}
          - clear_done
          - clear
        """
        action = str(body.get("action") or "").strip().lower()
        with SESSION_LOCK:
            if action == "add":
                resp = shopping_list.add(str(body.get("content") or ""))
            elif action == "remove":
                resp = shopping_list.remove(int(body.get("index") or 0))
            elif action == "toggle_done":
                resp = shopping_list.toggle_done(int(body.get("index") or 0))
            elif action == "clear_done":
                resp = shopping_list.clear_done()
            elif action == "clear":
                resp = shopping_list.clear()
            else:
                resp = ""
            items = shopping_list.to_list()
            pending = shopping_list.pending_count()
        if not resp:
            json_response(self, 200, {"ok": False, "error": f"未知操作：{action}"})
            return
        json_response(self, 200, {"ok": True, "response": resp,
                                  "items": items, "pending_count": pending})

/* =============== 购物清单（购买前需求池） =============== */
let listCache = [];

async function loadList(){
  const r = await api("/api/shopping_list","GET");
  renderList(r.items||[]);
}

function renderList(items){
  listCache = items;
  const pend = $("#list-pending"), done = $("#list-done");
  pend.innerHTML = ""; done.innerHTML = "";
  let np = 0, nd = 0;
  items.forEach((it, idx) => {
    const i = idx + 1;   // 编号=存储顺序，与对话指令「第N项」一致
    const row = document.createElement("div");
    row.style.cssText = "display:flex;align-items:center;gap:8px;padding:8px 0;border-bottom:1px solid var(--border);";
    row.innerHTML = `
      <div style="flex:1;min-width:0;">
        <div class="lc" style="${it.done ? "text-decoration:line-through;opacity:.6;" : ""}"></div>
        <div style="font-size:12px;opacity:.65;margin-top:2px;">#${i} · ${it.created_at}</div>
      </div>
      <button class="btn sm" onclick="goRecommend(${i})"><span class="material-symbols-outlined">forum</span> 去推荐</button>
      <button class="btn sm" onclick="listToggle(${i})">${it.done ? "取消完成" : "完成"}</button>
      <button class="btn sm danger" onclick="listRemove(${i})">移除</button>`;
    row.querySelector(".lc").textContent = it.content;   // 用户输入走 textContent 防注入
    (it.done ? done : pend).appendChild(row);
    it.done ? nd++ : np++;
  });
  if(!np) pend.innerHTML = '<div class="empty"><span class="big"><span class="material-symbols-outlined">checklist</span></span>暂无待处理需求 — 想到就记下来。</div>';
  $("#list-done-count").textContent = nd;
  $("#list-done-wrap").style.display = nd ? "" : "none";
}

async function listAction(action, extra={}){
  const r = await api("/api/shopping_list","POST",Object.assign({action}, extra));
  if(r.ok){ renderList(r.items||[]); updateListBadge(r.pending_count||0); }
}

function listAdd(){
  const v = ($("#list-input").value||"").trim();
  if(!v) return;
  listAction("add",{content:v});
  $("#list-input").value = "";
}
function listRemove(i){ listAction("remove",{index:i}); }
function listToggle(i){ listAction("toggle_done",{index:i}); }
function listClearDone(){ if(confirm("确认清空全部已完成条目？")) listAction("clear_done"); }
function listToggleDoneView(){ const d = $("#list-done"); d.style.display = d.style.display === "none" ? "" : "none"; }

// 去推荐：填入对话输入框并聚焦，不自动发送（抓取克制：由用户确认后再发）
function goRecommend(i){
  const it = listCache[i-1];
  if(!it) return;
  $("#chat-input").value = it.content;
  switchTab("tab-chat");
  $("#chat-input").focus();
}

function updateListBadge(n){
  const b = $("#list-badge");
  if(!b) return;
  b.textContent = n;
  b.style.display = n > 0 ? "" : "none";
}
