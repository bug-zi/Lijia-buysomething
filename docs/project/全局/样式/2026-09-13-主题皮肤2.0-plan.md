```python
# temp/check_contrast.py — 主题皮肤2.0 色板对比度校验（WCAG 相对亮度公式）
def lum(hx):
    hx = hx.lstrip('#'); r, g, b = (int(hx[i:i+2], 16)/255 for i in (0, 2, 4))
    f = lambda c: c/12.92 if c <= 0.04045 else ((c+0.055)/1.055)**2.4
    r, g, b = f(r), f(g), f(b)
    return 0.2126*r + 0.7152*g + 0.0722*b

def cr(a, b):
    la, lb = lum(a), lum(b)
    hi, lo = max(la, lb), min(la, lb)
    return (hi+0.05)/(lo+0.05)

# (皮肤, 模式, bg, card, text-1, text-2, text-3, primary, primary-dark)
P = [
 ("classic","light","#eef4fb","#fdfeff","#16283e","#41556f","#66788d","#1890ff","#096dd9"),
 ("classic","dark", "#0e1a2e","#16243a","#e9eef7","#b9c6da","#8798b1","#2e95ff","#9fd4ff"),
 ("sakura","light","#f9eff3","#fffcfd","#38202b","#5f4350","#836a75","#c06c84","#9a4e63"),
 ("sakura","dark", "#2a161f","#33202b","#f5e9ee","#d4bcc7","#a98e9b","#c87890","#eeb4c6"),
 ("forest","light","#eff6f1","#fcfefd","#1b2f25","#41584a","#647a6c","#3d8a68","#2c6650"),
 ("forest","dark", "#12211a","#1a2d23","#e8f1ec","#b9c8bd","#8a9c8e","#4a9a75","#a3d4bb"),
 ("sunset","light","#f9f2e8","#fffdf9","#33271a","#5a4b38","#7d6f5c","#cf8a3e","#a76a24"),
 ("sunset","dark", "#241a10","#2e2317","#f4ece0","#d4c5ad","#a5967f","#c27f35","#ecc690"),
]
fails = 0
for skin, mode, bg, card, t1, t2, t3, pri, prid in P:
    checks = [
        (f"text-1/bg",    cr(t1, bg),   7.0),
        (f"text-2/bg",    cr(t2, bg),   4.5),
        (f"text-3/bg",    cr(t3, bg),   3.0),
        (f"text-1/card",  cr(t1, card), 7.0),
        (f"white/primary",cr("#ffffff", pri), 3.0),   # 大字号实底按钮标准
    ]
    if mode == "light":
        checks.append(("primary-dark/card", cr(prid, card), 4.5))  # 代码/选中/链接文字（深色调）
    else:
        checks.append(("primary-dark(亮)/card", cr(prid, card), 4.5))  # 同上，深色下转为亮色调
    for name, v, need in checks:
        ok = v >= need
        fails += (not ok)
        print(f"{'PASS' if ok else 'FAIL'} {skin:7s} {mode:5s} {name:16s} {v:5.2f} (need {need})")
print(("ALL PASS" if not fails else f"{fails} FAILURES"))
raise SystemExit(1 if fails else 0)
```
